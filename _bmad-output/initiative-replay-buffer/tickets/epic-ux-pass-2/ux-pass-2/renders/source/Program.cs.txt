// Renders the ux-pass-2 design (story 10.1) in each theme. A scratch prototype, not the App: it applies
// the App's real theme dictionaries through ThemeService (by reflection) and composes each new screen
// from their styles and tokens, so the renders show the shipped look of every existing control.
// Usage: Render10 <fluent|gaming> <scene> <out.png>
// Scenes: library, library-search, welcome, audio-tracks, games, game-page, clips, hotkeys, storage, surfaces
public static class Program
{
    static bool G;
    static readonly List<Action<FrameworkElement, Canvas>> After = [];

    [STAThread]
    public static void Main(string[] a)
    {
        var theme = a[0];
        var scene = a[1];
        var outPath = a[2];
        G = theme == "gaming";
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        var asm = Assembly.Load("ReplayBuffer.App");
        var service = asm.GetType("ReplayBuffer.App.Theme.ThemeService")!
            .GetMethod("ForApplication", BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static)!
            .Invoke(null, [app])!;
        service.GetType().GetMethod("Apply")!.Invoke(service, [G ? AppTheme.GamingDark : AppTheme.Fluent]);

        app.Startup += (_, _) =>
        {
            var (content, height) = scene switch
            {
                "library" => (Library(selection: true), 600.0),
                "library-search" => (Library(selection: false), 600.0),
                "welcome" => (Welcome(), 600.0),
                "audio-tracks" => (AudioTracks(), 1000.0),
                "games" => (Games(), 600.0),
                "game-page" => (GamePage(), 1320.0),
                "clips" => (Clips(), 980.0),
                "hotkeys" => (Hotkeys(), 860.0),
                "storage" => (Storage(), 600.0),
                _ => (null, 0.0),
            };
            FrameworkElement root = content is null ? Surfaces() : (FrameworkElement)content;
            if (content is not null)
            {
                root = new Border { Width = 960, Height = height, Child = content, ClipToBounds = true };
                R((Border)root, Border.BackgroundProperty, "SolidBackgroundFillColorBaseBrush");
            }

            var overlay = new Canvas { IsHitTestVisible = false };
            var stack = new Grid();
            stack.Children.Add(root);
            stack.Children.Add(overlay);
            var w = new Window
            {
                Content = stack,
                SizeToContent = SizeToContent.WidthAndHeight,
                Left = -30000,
                Top = -30000,
                WindowStartupLocation = WindowStartupLocation.Manual,
                ShowActivated = false,
                ShowInTaskbar = false,
                WindowStyle = WindowStyle.None,
            };
            w.ContentRendered += (_, _) => w.Dispatcher.InvokeAsync(() =>
            {
                foreach (var act in After)
                {
                    act(stack, overlay);
                }

                stack.UpdateLayout();
                var rtb = new RenderTargetBitmap((int)(stack.ActualWidth * 2), (int)(stack.ActualHeight * 2), 192, 192, PixelFormats.Pbgra32);
                rtb.Render(stack);
                var enc = new PngBitmapEncoder();
                enc.Frames.Add(BitmapFrame.Create(rtb));
                using (var fs = File.Create(outPath))
                {
                    enc.Save(fs);
                }

                app.Shutdown();
            }, System.Windows.Threading.DispatcherPriority.ApplicationIdle);
            w.Show();
        };
        app.Run();
    }

    // ---------- helpers ----------

    public sealed record NavEntry(string Title, bool IsSection);

    static T R<T>(T e, DependencyProperty p, string key) where T : FrameworkElement
    {
        e.SetResourceReference(p, key);
        return e;
    }

    static TextBlock Tx(string s, double size = 14, FontWeight? w = null, string brush = "TextFillColorPrimaryBrush", bool wrap = false)
    {
        var t = new TextBlock
        {
            Text = s,
            FontSize = size,
            FontWeight = w ?? FontWeights.Normal,
            VerticalAlignment = VerticalAlignment.Center,
            TextWrapping = wrap ? TextWrapping.Wrap : TextWrapping.NoWrap,
            TextTrimming = wrap ? TextTrimming.None : TextTrimming.CharacterEllipsis,
        };
        return R(t, TextBlock.ForegroundProperty, brush);
    }

    static TextBlock Caption(string s, string brush = "TextFillColorSecondaryBrush") => Tx(s, 12, null, brush, wrap: true);

    static TextBlock Glyph(string g, double size = 16, string brush = "TextFillColorSecondaryBrush")
    {
        var t = Tx(g, size, null, brush);
        t.FontFamily = new FontFamily("Segoe Fluent Icons");
        return t;
    }

    static Border Box(UIElement? child, string bg, string? stroke = null, string radius = "ControlCornerRadius", Thickness? pad = null)
    {
        var b = new Border { Child = child, Padding = pad ?? new Thickness(12), BorderThickness = new Thickness(stroke is null ? 0 : 1) };
        R(b, Border.BackgroundProperty, bg);
        if (stroke is not null)
        {
            R(b, Border.BorderBrushProperty, stroke);
        }

        return R(b, Border.CornerRadiusProperty, radius);
    }

    static Border Card(UIElement child, Thickness? pad = null)
    {
        var b = new Border { Child = child, Padding = pad ?? new Thickness(16) };
        return R(b, FrameworkElement.StyleProperty, "RbCardStyle");
    }

    static Button Btn(string s, bool accent = false, double left = 0)
    {
        var b = new Button { Content = s, Margin = new Thickness(left, 0, 6, 0), VerticalAlignment = VerticalAlignment.Center };
        if (accent)
        {
            b.SetResourceReference(FrameworkElement.StyleProperty, "AccentButtonStyle");
        }

        return b;
    }

    static Button GlyphBtn(string glyph, string name)
    {
        var b = new Button { Content = Glyph(glyph, 14, "TextFillColorPrimaryBrush"), Padding = new Thickness(8, 6, 8, 6), VerticalAlignment = VerticalAlignment.Center, ToolTip = name };
        b.SetResourceReference(FrameworkElement.StyleProperty, "DefaultButtonStyle");
        return b;
    }

    static ComboBox Combo(string item, double width = 160)
    {
        var c = new ComboBox { MinWidth = width, Margin = new Thickness(0, 0, 12, 4), VerticalAlignment = VerticalAlignment.Center };
        c.Items.Add(item);
        c.SelectedIndex = 0;
        return c;
    }

    static ContentControl SectionHeader(string s)
    {
        var c = new ContentControl { Content = s, Focusable = false };
        return R(c, FrameworkElement.StyleProperty, "RbSectionHeaderStyle");
    }

    static FrameworkElement Section(string header, UIElement body, UIElement? right = null)
    {
        var p = new StackPanel { Margin = new Thickness(0, 0, 0, 20) };
        var h = new DockPanel { Margin = new Thickness(0, 0, 0, 8), LastChildFill = false };
        if (right is not null)
        {
            DockPanel.SetDock(right, Dock.Right);
            h.Children.Add(right);
        }

        h.Children.Add(SectionHeader(header));
        p.Children.Add(h);
        p.Children.Add(Card(body));
        return p;
    }

    static Grid Row(string label, UIElement control, string? help = null, double labelWidth = 170)
    {
        var g = new Grid { Margin = new Thickness(0, 4, 0, 4) };
        g.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(labelWidth) });
        g.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(1, GridUnitType.Star) });
        var l = Tx(label);
        l.VerticalAlignment = VerticalAlignment.Top;
        l.Margin = new Thickness(0, 6, 12, 0);
        g.Children.Add(l);
        var right = new StackPanel();
        right.Children.Add(control);
        if (help is not null)
        {
            var h = Caption(help);
            h.Margin = new Thickness(0, 4, 0, 0);
            right.Children.Add(h);
        }

        Grid.SetColumn(right, 1);
        g.Children.Add(right);
        return g;
    }

    static StackPanel H(params UIElement[] children)
    {
        var p = new StackPanel { Orientation = Orientation.Horizontal };
        foreach (var c in children)
        {
            p.Children.Add(c);
        }

        return p;
    }

    static StackPanel V(params UIElement[] children)
    {
        var p = new StackPanel();
        foreach (var c in children)
        {
            p.Children.Add(c);
        }

        return p;
    }

    static T M<T>(T e, double l, double t, double r, double b) where T : FrameworkElement
    {
        e.Margin = new Thickness(l, t, r, b);
        return e;
    }

    static TextBox Field(string text, double width) => new() { Text = text, Width = width, HorizontalAlignment = HorizontalAlignment.Left, VerticalAlignment = VerticalAlignment.Center };

    static ProgressBar Meter(double value, double width, double height = 6)
    {
        var p = new ProgressBar { Minimum = 0, Maximum = 1, Value = value, Width = width, Height = height, VerticalAlignment = VerticalAlignment.Center };
        return R(p, FrameworkElement.StyleProperty, "RbLevelMeterStyle");
    }

    static Border Pill(string s, bool accent)
    {
        var b = Box(Tx(s, 12, null, accent ? "AccentTextFillColorPrimaryBrush" : "TextFillColorSecondaryBrush"),
            accent ? "RbAccentSubtleBrush" : "ControlFillColorDefaultBrush", null, "RbSmallCornerRadius", new Thickness(6, 1, 6, 2));
        b.Margin = new Thickness(6, 0, 0, 0);
        b.VerticalAlignment = VerticalAlignment.Center;
        return b;
    }

    static Border Chip(bool changed)
    {
        var b = Box(Tx(changed ? "Changed for this game" : "Uses the main setting", 12, null, changed ? "AccentTextFillColorPrimaryBrush" : "TextFillColorSecondaryBrush"),
            changed ? "RbAccentSubtleBrush" : "ControlFillColorDefaultBrush", changed ? "AccentFillColorDefaultBrush" : "ControlStrokeColorDefaultBrush", "RbSmallCornerRadius", new Thickness(8, 2, 8, 3));
        b.VerticalAlignment = VerticalAlignment.Center;
        return b;
    }

    static Border KeyCap(string s) => M(Box(Tx(s, 14, FontWeights.SemiBold), "ControlFillColorDefaultBrush", "ControlStrokeColorDefaultBrush", "RbSmallCornerRadius", new Thickness(6, 1, 6, 2)), 4, 0, 4, 0);

    static TextBlock Link(string s)
    {
        var t = new TextBlock { VerticalAlignment = VerticalAlignment.Center, FontSize = 14 };
        var h = new Hyperlink(new Run(s));
        h.SetResourceReference(TextElement.ForegroundProperty, "AccentTextFillColorPrimaryBrush");
        t.Inlines.Add(h);
        return t;
    }

    static Border Flyout(UIElement child, double width)
    {
        var b = Box(child, "SolidBackgroundFillColorQuarternaryBrush", "SurfaceStrokeColorFlyoutBrush", "OverlayCornerRadius", new Thickness(4));
        b.Width = width;
        b.Effect = new System.Windows.Media.Effects.DropShadowEffect { BlurRadius = 24, ShadowDepth = 6, Opacity = 0.45, Color = Colors.Black };
        return b;
    }

    static UIElement MenuItemRow(string s, string? gesture = null, bool hover = false, bool check = false, bool disabled = false)
    {
        var g = new Grid { Height = 32, Margin = new Thickness(0, 1, 0, 1) };
        var hb = new Border();
        R(hb, Border.CornerRadiusProperty, "ControlCornerRadius");
        R(hb, Border.BackgroundProperty, hover ? "SubtleFillColorSecondaryBrush" : "SubtleFillColorTransparentBrush");
        g.Children.Add(hb);
        if (check)
        {
            g.Children.Add(M(Glyph("\uE73E", 12, "AccentTextFillColorPrimaryBrush"), 10, 0, 0, 0));
        }

        var t = Tx(s, 14, null, disabled ? "TextFillColorDisabledBrush" : "TextFillColorPrimaryBrush");
        t.Margin = new Thickness(32, 0, 12, 0);
        g.Children.Add(t);
        if (gesture is not null)
        {
            var k = Tx(gesture, 12, null, "TextFillColorSecondaryBrush");
            k.HorizontalAlignment = HorizontalAlignment.Right;
            k.Margin = new Thickness(0, 0, 12, 0);
            g.Children.Add(k);
        }

        return g;
    }

    static Border Separator()
    {
        var s = new Border { Height = 1, Margin = new Thickness(4) };
        return R(s, Border.BackgroundProperty, "DividerStrokeColorDefaultBrush");
    }

    static Border Dialog(string caption, string message, string? detail = null)
    {
        var dlg = new DockPanel { Width = 460 };
        var bar = H(Btn("Yes", accent: true), Btn("No"));
        bar.HorizontalAlignment = HorizontalAlignment.Right;
        foreach (Button b in bar.Children)
        {
            b.MinWidth = 96;
        }

        var barBox = Box(bar, "CardBackgroundFillColorSecondaryBrush", null, "ControlCornerRadius", new Thickness(24, 16, 18, 16));
        barBox.CornerRadius = new CornerRadius(0);
        barBox.BorderThickness = new Thickness(0, 1, 0, 0);
        R(barBox, Border.BorderBrushProperty, "DividerStrokeColorDefaultBrush");
        DockPanel.SetDock(barBox, Dock.Bottom);
        dlg.Children.Add(barBox);
        var body = new StackPanel { Margin = new Thickness(24, 28, 24, 24) };
        body.Children.Add(Tx(caption, 20, G ? FontWeights.Bold : FontWeights.SemiBold));
        body.Children.Add(M(Tx(message, 14, null, "TextFillColorPrimaryBrush", wrap: true), 0, 12, 0, 0));
        if (detail is not null)
        {
            body.Children.Add(M(Tx(detail, 14, null, "TextFillColorSecondaryBrush", wrap: true), 0, 8, 0, 0));
        }

        dlg.Children.Add(body);
        var box = Box(dlg, "SolidBackgroundFillColorBaseBrush", "SurfaceStrokeColorFlyoutBrush", "OverlayCornerRadius", new Thickness(0));
        box.ClipToBounds = true;
        return box;
    }

    // ---------- shell ----------

    static readonly string[] Pages = ["Library", "Settings", "General", "Clips", "Hotkeys", "Audio Tracks", "Games", "Sounds", "Storage and Cleanup"];

    static Grid Shell(string nav, UIElement page, string? title = null, UIElement? header = null, string state = "Ready to save Clips in Valorant", bool hint = true, string? dirtyNav = null)
    {
        var root = new Grid();
        root.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(200) });
        root.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(1, GridUnitType.Star) });

        var left = new DockPanel();
        left.SetResourceReference(FrameworkElement.MarginProperty, "RbNavPanePadding");
        var brand = new ContentControl { Content = "ReplayBuffer" };
        brand.SetResourceReference(FrameworkElement.MarginProperty, "RbNavBrandMargin");
        R(brand, FrameworkElement.StyleProperty, "RbBrandStyle");
        DockPanel.SetDock(brand, Dock.Top);
        left.Children.Add(brand);
        var list = new ListBox { BorderThickness = new Thickness(0), Padding = new Thickness(0) };
        R(list, Control.BackgroundProperty, "SubtleFillColorTransparentBrush");
        list.SetResourceReference(ItemsControl.ItemContainerStyleProperty, "RbNavItemStyle");
        list.ItemsSource = Pages.Select(p => new NavEntry(p, p == "Settings")).ToList();
        list.SelectedIndex = Array.IndexOf(Pages, nav);
        left.Children.Add(list);
        root.Children.Add(left);
        if (dirtyNav is not null)
        {
            After.Add((stack, canvas) =>
            {
                var item = (FrameworkElement)list.ItemContainerGenerator.ContainerFromIndex(Array.IndexOf(Pages, dirtyNav));
                var at = item.TranslatePoint(new Point(0, 0), stack);
                var dot = new Ellipse { Width = 6, Height = 6 };
                dot.SetResourceReference(Shape.FillProperty, "AccentFillColorDefaultBrush");
                if (G)
                {
                    dot = null!;
                }

                FrameworkElement mark = dot ?? (FrameworkElement)R(new Rectangle { Width = 6, Height = 6 }, Shape.FillProperty, "AccentFillColorDefaultBrush");
                Canvas.SetLeft(mark, at.X + 120);
                Canvas.SetTop(mark, at.Y + (item.ActualHeight / 2) - 3);
                canvas.Children.Add(mark);
            });
        }

        var layer = new Border();
        R(layer, FrameworkElement.StyleProperty, "RbShellContentLayerStyle");
        Grid.SetColumn(layer, 1);
        var col = new DockPanel();
        col.SetResourceReference(FrameworkElement.MarginProperty, "RbPagePadding");

        // Status header
        var dotEl = new Ellipse { Width = 10, Height = 10, Margin = new Thickness(0, 0, 8, 0), VerticalAlignment = VerticalAlignment.Center };
        dotEl.SetResourceReference(Shape.FillProperty, state.StartsWith("Waiting") ? "TextFillColorDisabledBrush" : "SystemFillColorSuccessBrush");
        var stateCc = new ContentControl { Content = state, VerticalAlignment = VerticalAlignment.Center };
        R(stateCc, FrameworkElement.StyleProperty, "RbStatusStateStyle");
        var hintText = Tx(hint ? "  ·  Alt+F10 saves the last 30 s" : "", 14, null, "TextFillColorSecondaryBrush");
        var status = Card(H(dotEl, stateCc, hintText), new Thickness(16, 12, 16, 12));
        DockPanel.SetDock(status, Dock.Top);
        col.Children.Add(status);

        FrameworkElement head;
        if (header is not null)
        {
            head = (FrameworkElement)header;
        }
        else
        {
            var t = new ContentControl { Content = title ?? nav };
            head = R(t, FrameworkElement.StyleProperty, "RbPageTitleStyle");
        }

        head.Margin = new Thickness(0, 16, 0, 0);
        DockPanel.SetDock(head, Dock.Top);
        col.Children.Add(head);
        var pageHost = new ContentControl { Content = page, Margin = new Thickness(0, 12, 0, 0), Focusable = false };
        col.Children.Add(pageHost);
        layer.Child = col;
        root.Children.Add(layer);
        return root;
    }

    static Border LibRow(string name, string meta, string length, bool selected = false, bool focused = false, bool fav = false, bool saving = false, string? kind = null)
    {
        var grid = new Grid();
        grid.ColumnDefinitions.Add(new ColumnDefinition { Width = GridLength.Auto });
        grid.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(1, GridUnitType.Star) });
        grid.ColumnDefinitions.Add(new ColumnDefinition { Width = GridLength.Auto });
        var thumb = new Grid();
        var tb = R(new Border(), FrameworkElement.StyleProperty, "RbLibraryThumbnailStyle");
        tb.Child = R(new System.Windows.Shapes.Path(), FrameworkElement.StyleProperty, "RbLibraryPlaceholderGlyphStyle");
        thumb.Children.Add(tb);
        var badge = R(new Border(), FrameworkElement.StyleProperty, "RbLibraryLengthBadgeStyle");
        badge.Child = R(new TextBlock { Text = length }, FrameworkElement.StyleProperty, "RbLibraryLengthTextStyle");
        thumb.Children.Add(badge);
        if (kind is not null)
        {
            var kb = R(new Border(), FrameworkElement.StyleProperty, "RbLibraryLengthBadgeStyle");
            kb.HorizontalAlignment = HorizontalAlignment.Left;
            kb.VerticalAlignment = VerticalAlignment.Top;
            kb.Child = R(new TextBlock { Text = kind }, FrameworkElement.StyleProperty, "RbLibraryLengthTextStyle");
            thumb.Children.Add(kb);
        }

        grid.Children.Add(thumb);
        var text = new StackPanel { Margin = new Thickness(12, 0, 12, 0), VerticalAlignment = VerticalAlignment.Center };
        text.Children.Add(R(new TextBlock { Text = name }, FrameworkElement.StyleProperty, "RbLibraryNameStyle"));
        text.Children.Add(R(new TextBlock { Text = meta }, FrameworkElement.StyleProperty, "RbLibraryMetaStyle"));
        if (saving)
        {
            var pill = R(new Border { HorizontalAlignment = HorizontalAlignment.Left }, FrameworkElement.StyleProperty, "RbLibraryLockPillStyle");
            pill.Child = R(new TextBlock { Text = "Saving…" }, FrameworkElement.StyleProperty, "RbLibraryLockTextStyle");
            text.Children.Add(pill);
        }

        Grid.SetColumn(text, 1);
        grid.Children.Add(text);
        var star = new ToggleButton { IsChecked = fav, ToolTip = "Favourite" };
        R(star, FrameworkElement.StyleProperty, "RbStarToggleStyle");
        Grid.SetColumn(star, 2);
        grid.Children.Add(star);

        var row = R(new Border { Child = grid }, FrameworkElement.StyleProperty, "RbLibraryRowStyle");
        if (selected)
        {
            R(row, Border.BackgroundProperty, "RbAccentSubtleBrush");
        }

        var outer = new Grid();
        outer.Children.Add(row);
        if (selected)
        {
            var bar = new Border { Width = 3, HorizontalAlignment = HorizontalAlignment.Left, Margin = new Thickness(0, 8, 0, 8), CornerRadius = new CornerRadius(G ? 0 : 1.5) };
            R(bar, Border.BackgroundProperty, "AccentFillColorDefaultBrush");
            outer.Children.Add(bar);
        }

        var wrap = new Border { Child = outer, BorderThickness = new Thickness(focused ? 2 : 0), Padding = new Thickness(focused ? 0 : 2), Margin = new Thickness(0, 0, 0, 0) };
        if (focused)
        {
            R(wrap, Border.BorderBrushProperty, "FocusStrokeColorOuterBrush");
            R(wrap, Border.CornerRadiusProperty, "OverlayCornerRadius");
        }

        return wrap;
    }
