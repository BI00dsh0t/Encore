// Renders the 9.1 design (DESIGN.md / EXPERIENCE.md) in each theme. Scratch prototype, not the App.
// Usage: Render <fluent-dark|fluent-light|gaming> <library|settings|surfaces> <out.png>
public static class Program
{
    static bool G;
    static string U(string s) => G ? s.ToUpperInvariant() : s;

    [STAThread]
    public static void Main(string[] a)
    {
        var theme = a[0]; var scene = a[1]; var outPath = a[2];
        G = theme == "gaming";
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.ThemeMode = theme == "fluent-light" ? ThemeMode.Light : ThemeMode.Dark;
        var dir = AppContext.BaseDirectory;
        app.Resources.MergedDictionaries.Add(Load(System.IO.Path.Combine(dir, G ? "gaming.xaml.txt" : "fluent-app-keys.xaml.txt")));
        app.Startup += (_, _) =>
        {
            FrameworkElement root = scene switch { "library" => Shell(true), "settings" => Shell(false), _ => Surfaces() };
            var w = new Window { Content = root, SizeToContent = SizeToContent.WidthAndHeight, Left = -30000, Top = -30000, WindowStartupLocation = WindowStartupLocation.Manual, ShowActivated = false, ShowInTaskbar = false };
            w.ContentRendered += (_, _) => w.Dispatcher.InvokeAsync(() =>
            {
                var rtb = new RenderTargetBitmap((int)(root.ActualWidth * 2), (int)(root.ActualHeight * 2), 192, 192, PixelFormats.Pbgra32);
                rtb.Render(root);
                var enc = new PngBitmapEncoder(); enc.Frames.Add(BitmapFrame.Create(rtb));
                using (var fs = File.Create(outPath)) enc.Save(fs);
                app.Shutdown();
            }, System.Windows.Threading.DispatcherPriority.ApplicationIdle);
            w.Show();
        };
        app.Run();
    }

    static ResourceDictionary Load(string path) { using var s = File.OpenRead(path); return (ResourceDictionary)XamlReader.Load(s); }

    static T R<T>(T e, DependencyProperty p, string key) where T : FrameworkElement { e.SetResourceReference(p, key); return e; }

    static TextBlock Tx(string s, double size = 14, FontWeight? w = null, string brush = "TextFillColorPrimaryBrush", double spacing = 0)
    {
        var t = new TextBlock { Text = s, FontSize = size, FontWeight = w ?? FontWeights.Normal, TextTrimming = TextTrimming.CharacterEllipsis, VerticalAlignment = VerticalAlignment.Center };
        if (G) t.FontFamily = new FontFamily("Segoe UI");
        return R(t, TextBlock.ForegroundProperty, brush);
    }

    static Border Box(UIElement? child, string bg, string? stroke = null, string radius = "OverlayCornerRadius", Thickness? pad = null)
    {
        var b = new Border { Child = child, Padding = pad ?? new Thickness(16), BorderThickness = new Thickness(stroke is null ? 0 : 1) };
        R(b, Border.BackgroundProperty, bg);
        if (stroke is not null) R(b, Border.BorderBrushProperty, stroke);
        return R(b, Border.CornerRadiusProperty, radius);
    }

    static Border Card(UIElement child, Thickness? pad = null) => Box(child, "CardBackgroundFillColorDefaultBrush", "CardStrokeColorDefaultBrush", "OverlayCornerRadius", pad);

    static TextBlock SectionHeader(string s) => G
        ? Tx(s.ToUpperInvariant(), 12, FontWeights.Bold, "TextFillColorSecondaryBrush")
        : Tx(s, 14, FontWeights.SemiBold);

    static TextBlock PageTitle(string s) => G ? Tx(s.ToUpperInvariant(), 26, FontWeights.Black) : Tx(s, 28, FontWeights.SemiBold);

    static UIElement NavItem(string s, bool selected)
    {
        var g = new Grid { Height = G ? 40 : 36, Margin = new Thickness(G ? 0 : 4, 0, G ? 0 : 4, 2) };
        var bg = new Border();
        R(bg, Border.CornerRadiusProperty, "RbNavItemCornerRadius");
        R(bg, Border.BackgroundProperty, selected ? (G ? "RbAccentSubtleBrush" : "SubtleFillColorSecondaryBrush") : "SubtleFillColorTransparentBrush");
        g.Children.Add(bg);
        if (selected)
        {
            var bar = new Border { Width = 3, HorizontalAlignment = HorizontalAlignment.Left, Height = G ? double.NaN : 16, VerticalAlignment = G ? VerticalAlignment.Stretch : VerticalAlignment.Center, CornerRadius = new CornerRadius(G ? 0 : 1.5) };
            R(bar, Border.BackgroundProperty, "AccentFillColorDefaultBrush");
            g.Children.Add(bar);
        }
        var t = G ? Tx(s.ToUpperInvariant(), 13, FontWeights.SemiBold, selected ? "AccentTextFillColorPrimaryBrush" : "TextFillColorSecondaryBrush")
                  : Tx(s, 14, FontWeights.Normal);
        t.Margin = new Thickness(G ? 16 : 12, 0, 8, 0);
        g.Children.Add(t);
        return g;
    }

    static FrameworkElement Shell(bool library)

        var root = new Grid { Width = 900, Height = 600 };
        R(root, Panel.BackgroundProperty, "SolidBackgroundFillColorBaseBrush");
        root.ColumnDefinitions.Add(new ColumnDefinition { Width = new GridLength(200) });

        var nav = new StackPanel { Margin = new Thickness(0, 16, 0, 16) };
        var brand = G ? Tx("REPLAYBUFFER", 15, FontWeights.Black) : Tx("ReplayBuffer", 15, FontWeights.SemiBold);

        label.Margin = new Thickness(16, 14, 0, 6);
        nav.Children.Add(label);
        foreach (var p in new[] { "General", "Replay Buffer", "Hotkeys", "Track Layout", "Game Profiles", "Feedback Sounds", "Recordings", "Library and Cleanup" })

        root.Children.Add(nav);

        // Content layer: Fluent's NavigationView layer over Mica; Gaming shares bg-base.
        var layer = new Border { BorderThickness = new Thickness(G ? 0 : 1, G ? 0 : 1, 0, 0), CornerRadius = new CornerRadius(G ? 0 : 8, 0, 0, 0) };
        R(layer, Border.BackgroundProperty, "LayerFillColorDefaultBrush");
        R(layer, Border.BorderBrushProperty, "CardStrokeColorDefaultBrush");
        Grid.SetColumn(layer, 1);
        root.Children.Add(layer);

        var main = new DockPanel { Margin = new Thickness(24, 20, 24, 16) };
        layer.Child = main;

        var status = new StackPanel();
        var line = new StackPanel { Orientation = Orientation.Horizontal };
        var dot = new Ellipse { Width = 10, Height = 10, Margin = new Thickness(0, 0, 10, 0), VerticalAlignment = VerticalAlignment.Center };

        line.Children.Add(Tx(U("Buffering on Valorant · Mic muted"), 14, G ? FontWeights.Bold : FontWeights.SemiBold));
        status.Children.Add(line);
        var warn = new DockPanel { Margin = new Thickness(20, 8, 0, 0) };
        var dismiss = new Button { Content = U("Dismiss"), Padding = new Thickness(10, 2, 10, 2) };
        DockPanel.SetDock(dismiss, Dock.Right);

        var sev = Tx("Warning", 14, FontWeights.SemiBold, "SystemFillColorCautionBrush"); sev.Margin = new Thickness(0, 0, 8, 0);

        DockPanel.SetDock(statusCard, Dock.Top);
        main.Children.Add(statusCard);

        if (library) LibraryPage(main); else GeneralPage(main);
        return root;
    }

    static void LibraryPage(DockPanel main)
    {
        var header = new DockPanel { Margin = new Thickness(0, 0, 0, 12) };

        void Combo(string label, string value, double width)
        {
            var l = Tx(label, 12, null, "TextFillColorSecondaryBrush"); l.Margin = new Thickness(0, 0, 6, 0);
            filters.Children.Add(l);

            filters.Children.Add(c);
        }
        Combo("Group by", "Game", 90); Combo("Game", "All games", 120); Combo("Date", "All dates", 110);
        filters.Children.Add(new ToggleButton { Content = U("★ Favourites only"), IsChecked = false, Padding = new Thickness(10, 4, 10, 4) });

        header.Children.Add(title);
        header.Children.Add(filters);
        DockPanel.SetDock(header, Dock.Top);

        {
            var g = new DockPanel { Margin = new Thickness(0, 6, 0, 6) };

            R(rule, Border.BackgroundProperty, "DividerStrokeColorDefaultBrush");
            g.Children.Add(rule);
            list.Children.Add(g);
        }
        void Row(string name, string meta, string len, Color a, Color b, bool fav, bool saving = false, bool hover = false)
        {
            var row = new DockPanel();
            var thumb = new Grid { Width = 128, Height = 72 };
            var img = new Border { Background = new LinearGradientBrush(a, b, 35) }; // stands in for the real thumbnail
            R(img, Border.CornerRadiusProperty, "RbSmallCornerRadius");

            badge.HorizontalAlignment = HorizontalAlignment.Right; badge.VerticalAlignment = VerticalAlignment.Bottom; badge.Margin = new Thickness(6);

            row.Children.Add(thumb);

            star.Template = StarTemplate();
            star.SetResourceReference(Control.ForegroundProperty, fav ? "AccentTextFillColorPrimaryBrush" : "TextFillColorSecondaryBrush");

            row.Children.Add(star);

            text.Children.Add(Tx(name, 14, G ? FontWeights.Bold : FontWeights.SemiBold));

            {
                var pill = Box(Tx("Saving…", 12, null, "SystemFillColorCautionBrush"), "SystemFillColorCautionBackgroundBrush", null, "RbSmallCornerRadius", new Thickness(6, 1, 6, 1));
                pill.HorizontalAlignment = HorizontalAlignment.Left; pill.Margin = new Thickness(0, 6, 0, 0);
                text.Children.Add(pill);
            }
            row.Children.Add(text);
            var card = Box(row, hover ? "ControlFillColorSecondaryBrush" : "CardBackgroundFillColorDefaultBrush", "CardStrokeColorDefaultBrush", "OverlayCornerRadius", new Thickness(8));
            card.Margin = new Thickness(0, 0, 0, 6);
            list.Children.Add(card);
        }
        Group("Valorant");
        Row("Valorant_2026-11-04_21-14-02", "Today 21:14 · 182 MB", "0:30", Color.FromRgb(0xC8, 0x3B, 0x4A), Color.FromRgb(0x3A, 0x1C, 0x4F), true, saving: true);

    }

    static ControlTemplate StarTemplate()
    {
        var f = new FrameworkElementFactory(typeof(Border));
        f.SetResourceReference(Border.CornerRadiusProperty, "ControlCornerRadius");
        f.SetValue(Border.BackgroundProperty, Brushes.Transparent);
        var cp = new FrameworkElementFactory(typeof(ContentPresenter));
        cp.SetValue(FrameworkElement.HorizontalAlignmentProperty, HorizontalAlignment.Center);
        cp.SetValue(FrameworkElement.VerticalAlignmentProperty, VerticalAlignment.Center);
        f.AppendChild(cp);
        return new ControlTemplate(typeof(ToggleButton)) { VisualTree = f };
    }
